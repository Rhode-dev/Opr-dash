Cloud Sync stores one snapshot of the current manifest in your own Supabase project. Upload from the PC, then Download on the phone (or enable auto-download) to continue with the same containers. Training is not included in the snapshot.

You stay signed in on this device — Supabase refreshes the session automatically. You only need to sign in again if the session is revoked or expires.
One-time Supabase setup (table + security)
1. Create a project at supabase.com/dashboard.
2. Open SQL Editor, paste and run the SQL below.
3. Open Project Settings → API and copy the Project URL and anon public key into the fields here.
4. Create an account here with an email and password, then sign in on the other device with the same account.
If Supabase asks for email confirmation, confirm the email first or turn confirmation off in Authentication settings. 
create table if not exists public.manifest_snapshots (
  user_id uuid primary key references auth.users(id) on delete cascade,
  payload jsonb not null,
  updated_at timestamptz not null default now()
);
alter table public.manifest_snapshots enable row level security;
drop policy if exists "own manifest snapshot" on public.manifest_snapshots;
create policy "own manifest snapshot"
  on public.manifest_snapshots
  for all
  to authenticated
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);

One-time team SQL setup
Run this in the same Supabase project as Personal sync. It creates team workspaces, members, roles, snapshots, approvals and helper functions.
create table if not exists public.team_workspaces (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  owner uuid not null references auth.users(id) on delete cascade,
  join_code text unique not null,
  approval_required boolean not null default false,
  created_at timestamptz not null default now()
);
create table if not exists public.team_members (
  workspace_id uuid references public.team_workspaces(id) on delete cascade,
  user_id uuid references auth.users(id) on delete cascade,
  email text,
  role text not null check (role in ('viewer','editor','admin')),
  joined_at timestamptz not null default now(),
  primary key (workspace_id,user_id)
);
create table if not exists public.team_snapshots (
  workspace_id uuid primary key references public.team_workspaces(id) on delete cascade,
  payload jsonb not null,
  revision bigint not null default 1,
  updated_at timestamptz not null default now(),
  updated_by uuid
);
create table if not exists public.team_change_requests (
  id uuid primary key default gen_random_uuid(),
  workspace_id uuid references public.team_workspaces(id) on delete cascade,
  user_id uuid references auth.users(id) on delete cascade,
  email text,
  payload jsonb not null,
  note text,
  status text not null default 'pending' check (status in ('pending','approved','rejected')),
  created_at timestamptz not null default now(),
  reviewed_by uuid,
  reviewed_at timestamptz
);
alter table public.team_workspaces enable row level security;
alter table public.team_members enable row level security;
alter table public.team_snapshots enable row level security;
alter table public.team_change_requests enable row level security;
create or replace function public.team_role(ws uuid) returns text
language sql security definer set search_path=public stable as $$
  select role from team_members where workspace_id=ws and user_id=auth.uid()
$$;
create policy "team workspaces read" on public.team_workspaces for select to authenticated using (public.team_role(id) is not null);
create policy "team workspaces update" on public.team_workspaces for update to authenticated using (public.team_role(id)='admin');
create policy "team members read" on public.team_members for select to authenticated using (public.team_role(workspace_id) is not null);
create policy "team members admin" on public.team_members for update to authenticated using (public.team_role(workspace_id)='admin') with check (public.team_role(workspace_id)='admin');
create policy "team members delete" on public.team_members for delete to authenticated using (public.team_role(workspace_id)='admin');
create policy "team snapshots read" on public.team_snapshots for select to authenticated using (public.team_role(workspace_id) is not null);
create policy "team snapshots write" on public.team_snapshots for insert to authenticated with check (public.team_role(workspace_id) in ('editor','admin'));
create policy "team snapshots update" on public.team_snapshots for update to authenticated using (public.team_role(workspace_id) in ('editor','admin'));
create policy "team requests read" on public.team_change_requests for select to authenticated using (public.team_role(workspace_id) in ('editor','admin') and (user_id=auth.uid() or public.team_role(workspace_id)='admin'));
create policy "team requests insert" on public.team_change_requests for insert to authenticated with check (user_id=auth.uid() and public.team_role(workspace_id) in ('editor','admin'));
create policy "team requests update" on public.team_change_requests for update to authenticated using (public.team_role(workspace_id)='admin');
create policy "team members leave" on public.team_members for delete to authenticated using (user_id=auth.uid());
create or replace function public.create_team(p_name text,p_code text) returns uuid
language plpgsql security definer set search_path=public as $$
declare ws uuid;
begin
  insert into team_workspaces(name,owner,join_code) values(p_name,auth.uid(),p_code) returning id into ws;
  insert into team_members(workspace_id,user_id,email,role) values(ws,auth.uid(),auth.email(),'admin');
  insert into team_snapshots(workspace_id,payload,revision) values(ws,'{}'::jsonb,1);
  return ws;
end $$;
create or replace function public.join_team(p_code text) returns uuid
language plpgsql security definer set search_path=public as $$
declare ws uuid;
begin
  select id into ws from team_workspaces where join_code=p_code;
  if ws is null then raise exception 'Invalid join code'; end if;
  insert into team_members(workspace_id,user_id,email,role) values(ws,auth.uid(),auth.email(),'editor')
  on conflict (workspace_id,user_id) do nothing;
  return ws;
end $$;
grant execute on function public.team_role(uuid) to authenticated;
grant execute on function public.create_team(text,text) to authenticated;
grant execute on function public.join_team(text) to authenticated;
grant select, insert, update, delete on public.team_workspaces to authenticated;
grant select, insert, update, delete on public.team_members to authenticated;
grant select, insert, update, delete on public.team_snapshots to authenticated;
grant select, insert, update, delete on public.team_change_requests to authenticated;

One-time share SQL 
create table if not exists public.shared_snapshots (
  id uuid primary key default gen_random_uuid(),
  token text unique not null,
  payload jsonb not null,
  created_by uuid references auth.users(id) on delete cascade,
  created_at timestamptz not null default now(),
  expires_at timestamptz not null
);
alter table public.shared_snapshots enable row level security;
create policy "share insert own" on public.shared_snapshots for insert to authenticated with check (created_by=auth.uid());
create or replace function public.get_shared_snapshot(p_token text) returns jsonb
language sql security definer set search_path=public stable as $$
  select payload from shared_snapshots where token=p_token and expires_at>now()
$$;
revoke all on public.shared_snapshots from anon;
grant execute on function public.get_shared_snapshot(text) to anon, authenticated;
Anyone with the link can view that snapshot until it expires. It is read-only and does not expose your Supabase account.
